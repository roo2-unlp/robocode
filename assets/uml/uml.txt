@startuml
skinparam classAttributeIconSize 0

interface BulletEffect {
    +applyEffect(bullet: BulletPeer, target: RobotPeer)
}

class NoEffect {
    +<<create>>NoEffect(): NoEffect
    +applyEffect(bullet: BulletPeer, target: RobotPeer)
}

class StunningEffect {
    + stunDuration: int
    +<<create>>StunningEffect(stunDuration: int): StunningEffect
    +applyEffect(bullet: BulletPeer, target: RobotPeer)
}

class BattleRules {
    + bulletEffect : String
    + stunDuration : int
    + getBulletEffect() : String
    + getStunDuration(): int
}

class BulletPeer {
    + checkRobotCollision()
}

class RobotPeer {
    + skipNextTurns(turns: int)
    + getBulletEffect() : BulletEffect
}

class BulletEffectCreator {
    + {static} <<create>> createBulletEffect(battleRules: BattleRules): BulletEffect
}
RobotPeer *-- BulletPeer
RobotPeer --* BulletEffect
RobotPeer --> BulletEffectCreator
BulletEffect <|-- NoEffect
BulletEffect <|-- StunningEffect
RobotPeer --* BattleRules

note right of BulletPeer::checkRobotCollision
//Apply the bullet effect
this.owner.getBulletEffect()
    .applyEffect(this, otherRobot);
end note
@enduml
